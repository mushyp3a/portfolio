{{- /* Returns ONLY the resolved image URL for a photograph (no tag), so the lightbox can drop it into its own markup. Resolution rules are identical to photo-image.html — Immich takes precedence over the (now unused) placeholder. Whitespace is trimmed so the result is a clean URL string. Quality tiers (lowest to highest): thumbnail -> immichBaseURL (256px, lightest) preview -> immichPreviewURL (~1440p, light) fullsize -> immichFullsizeURL (DEFAULT: full-res web JPEG, sharp) original -> immichOriginalURL (raw/file; lightbox + single page) site param immichUseOriginal = true forces the grid to originals too (heaviest load). The lightbox always requests "original" explicitly. */ -}} {{- $quality := .quality | default "preview" -}} {{- if site.Params.immichUseOriginal }}{{ $quality = "original" }}{{ end -}} {{- $base := site.Params.immichBaseURL -}} {{- with site.Params.immichPreviewURL }}{{ if eq $quality "preview" }}{{ $base = . }}{{ end }}{{ end -}} {{- with site.Params.immichFullsizeURL }}{{ if eq $quality "fullsize" }}{{ $base = . }}{{ end }}{{ end -}} {{- with site.Params.immichOriginalURL }}{{ if eq $quality "original" }}{{ $base = . }}{{ end }}{{ end -}} {{- $id := .page.Params.immichId -}} {{- $src := .page.Params.image | default "" -}} {{- if and $id $base }}{{ $src = replace $base "{id}" $id }}{{ end -}} {{- $src -}}
